function doOptions(e) {
  // Native GAS CORS response. Do NOT use .setHeaders()
  return ContentService.createTextOutput("")
    .setMimeType(ContentService.MimeType.JSON);
}

function createJsonResponse(responseObject) {
  return ContentService.createTextOutput(JSON.stringify(responseObject))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  const params = e.parameter || {};
  
  try {
    const action = params.action || params.action_admin;
    
    if (!action) {
      return createJsonResponse({ status: "error", message: "Missing directive action parameter." });
    }
    
    // Website Authentication Handlers
    if (action === "LOGIN_USER") {
      return createJsonResponse(processWebsiteLogin(params));
    }
    if (action === "REGISTER_USER") {
      return createJsonResponse(processWebsiteRegistration(params));
    }
    
    // Webapp Polling Queries
    if (action === "GET_USERS") {
      return handleGetUsers();
    }
    if (action === "GET_APPLICATIONS") {
      return handleGetApplications();
    }
    
    return createJsonResponse({ status: "error", message: "Unrecognized GET target." });
  } catch (err) {
    return createJsonResponse({ status: "error", message: err.toString() });
  }
}

function doPost(e) {
  try {
    if (!e.postData || !e.postData.contents) {
      throw new Error("Null payload submitted.");
    }
    
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;
    const data = payload.data || {};
    
    switch(action) {
      // Telemetry & Economy
      case "ADD_JOB":             return handleAddJob(data);
      case "UPDATE_CARGO_RATE":   return handleUpdateCargoRate(data);

      // Identity & Permissions
      case "LOGIN_USER":          return handleLoginUser(data);
      case "REGISTER_USER":       return handleRegisterUser(data);
      case "GET_USERS":           return handleGetUsers();
      case "APPROVE_USER":        return handleApproveUser(data);
      case "MANAGE_USER":         return handleManageUser(data);

      // Event Matrix
      case "ADD_EVENT":           return handleAddEvent(data);
      case "UPDATE_EVENT":        return handleUpdateEvent(data);
      case "DELETE_EVENT":        return handleDeleteEvent(data);

      // Tour Infrastructure
      case "CREATE_TOUR":         return handleCreateTour(data);
      case "UPDATE_TOUR_STATUS":  return handleUpdateTourStatus(data);
      case "UPDATE_TOUR_MASTER":  return handleUpdateTourMaster(data);
      case "DELETE_TOUR":         return handleDeleteTour(data);

      // Media & Intel Management
      case "UPDATE_ASSET":        return handleUpdateAsset(data);
      case "ADD_NEWS":            return handleAddNews(data);
      case "DELETE_NEWS":         return handleDeleteNews(data);
      case "ADD_GALLERY":         return handleAddGallery(data);
      case "DELETE_GALLERY":      return handleDeleteGallery(data);
      case "SUBMIT_APPLICATION":  return handleSubmitApplication(data);

      default:
        throw new Error("Invalid command directive: " + action);
    }
  } catch (error) {
    return createJsonResponse({ status: "error", message: error.message });
  }
}

function getTargetSheet(sheetName) {
  let targetId = "";
  
  if (["Site_Assets", "Website_News", "Website_Gallery", "APPLICATIONS", "USERS_DB"].includes(sheetName)) {
    targetId = DB_WEBSITE;
  } else if (["Live_Job_Logs"].includes(sheetName)) {
    targetId = DB_JOBS;
  } else if (["Cargo_Rates"].includes(sheetName)) {
    targetId = CARGO_SPREADSHEET_ID;
  } else if (["ENTRY SHEET", "EVENT_RECORDS"].includes(sheetName)) {
    targetId = DB_EVENTS;
  } else {
    // Falls back to Campaigns/Tours Spreadsheet for dynamic tables
    targetId = DB_TOURS;
  }
  
  try {
    const ss = SpreadsheetApp.openById(targetId);
    let sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    }
    return { sheet: sheet, ss: ss };
  } catch (e) {
    throw new Error(`DB_ROUTER_FAIL: Target ID [${targetId}] for Table [${sheetName}]: ${e.message}`);
  }
}

function extractSafeNumber(val) {
  if (val === null || val === undefined) return 0;
  // Convert to string, strip everything except numbers and decimal point
  const cleanStr = String(val).replace(/[^0-9.]/g, '');
  const parsed = parseFloat(cleanStr);
  return isNaN(parsed) ? 0 : parsed;
}

function handleAddJob(data) {
  const cargoSs = SpreadsheetApp.openById(CARGO_SPREADSHEET_ID);
  const cargoRatesMap = getCargoRatesMap(cargoSs);
  const { sheet: jobSheet } = getTargetSheet("Live_Job_Logs");

  const cargoName = data.cargo || "General Cargo";
  const rawMass = extractSafeNumber(data.mass);
  
  // If the tracker emits in KG (e.g., 20000kg), adjust to Tonnes. If <= 100, assume Tonnes.
  const tonnes = rawMass > 100 ? (rawMass / 1000) : rawMass; 
  const distance = extractSafeNumber(data.driven_distance);
  const damagePercentage = extractSafeNumber(data.cargo_damage);

  // Dynamic Payout Fix
  const finalCalculatedRevenue = calculateJobRevenue(cargoName, tonnes, distance, damagePercentage, cargoRatesMap);

  // Initialize Sheet Layout if empty
  if (jobSheet.getLastRow() === 0) {
    jobSheet.appendRow([
      "Timestamp", "Job ID", "Driver", "Cargo", "Weight", 
      "Source City", "Source Co", "Dest City", "Dest Co", 
      "Truck", "Trailer", "Planned Dist", "Driven Dist", 
      "Top Speed", "Damage %", "Revenue (€)", "Auto Park", 
      "Fuel Used", "Duration", "Avg Speed", "Timeline"
    ]);
    jobSheet.setFrozenRows(1);
  }

  const newRow = [
    Utilities.formatDate(new Date(), "GMT+05:30", "dd-MMM-yyyy hh:mm a"),
    data.job_id || Utilities.getUuid().split('-')[0].toUpperCase(),
    data.driver_name || "UNKNOWN",
    cargoName,
    tonnes.toFixed(1) + " t",
    data.source_city || "Unknown",
    data.source_company || "Unknown",
    data.dest_city || "Unknown",
    data.dest_company || "Unknown",
    data.truck || "Fleet Standard",
    data.trailer || "Default Platform",
    extractSafeNumber(data.planned_distance) || distance,
    distance,
    extractSafeNumber(data.top_speed),
    damagePercentage,
    finalCalculatedRevenue, // Overwritten dynamically with safe math
    data.auto_park ? "YES" : "NO",
    extractSafeNumber(data.fuel_used),
    data.duration || "0h 0m",
    extractSafeNumber(data.avg_speed),
    data.timeline || "Clean Drive"
  ];

  jobSheet.appendRow(newRow);

  return createJsonResponse({ 
    status: "success", 
    calculated_revenue: finalCalculatedRevenue,
    distance_logged: distance,
    tonnes_evaluated: tonnes
  });
}

function getCargoRatesMap(ss) {
  const cargoMap = new Map();
  const sheet = ss.getSheetByName("Cargo_Rates");
  
  if (!sheet) return cargoMap;

  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    const rawCargoName = data[i][0];
    const rawBasePrice = data[i][4]; // Index 4 == Column E
    
    if (rawCargoName) {
      const cargoKey = String(rawCargoName).trim().toUpperCase();
      const basePrice = extractSafeNumber(rawBasePrice);
      
      // CRITICAL: Do not add to map if the price is 0, empty, or invalid
      if (basePrice > 0) {
        cargoMap.set(cargoKey, basePrice);
      }
    }
  }
  return cargoMap;
}

function calculateJobRevenue(cargoName, tonnes, distance, damagePercentage, cargoRatesMap) {
  const DEFAULT_BASE_RATE = 6.50; // Fallback unit tariff
  const MAX_CAPPED_DISTANCE = 3000; // Prevent macro glitches
  
  const safeCargoKey = String(cargoName || "").trim().toUpperCase();
  const safeTonnes = Math.max(0.1, extractSafeNumber(tonnes)); // Ensure at least 0.1t to prevent *0 wiping out revenue
  const rawDist = extractSafeNumber(distance);
  const cappedDistance = Math.min(rawDist, MAX_CAPPED_DISTANCE);
  const safeDamage = Math.min(100, Math.max(0, extractSafeNumber(damagePercentage)));

  let baseRate = cargoRatesMap.has(safeCargoKey) ? cargoRatesMap.get(safeCargoKey) : 0;
  
  // CRITICAL: Absolute fallback. Never allow baseRate to be 0 or undefined.
  if (!baseRate || baseRate <= 0) {
    baseRate = DEFAULT_BASE_RATE;
  }

  // Revenue Engine: Gross Value = Base Rate * Tonnes * Capped Distance
  const grossValue = baseRate * safeTonnes * cappedDistance;
  
  // Damage Penalty: Deduct linearly proportional to the load's integrity loss
  const penalty = grossValue * (safeDamage / 100);
  const finalPayout = Math.max(0, Math.round(grossValue - penalty));

  return finalPayout;
}

function handleUpdateCargoRate(data) {
  const { sheet } = getTargetSheet("Cargo_Rates");
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Cargo Name", "In-Game Category", "Weight", "Required DLC", "Base Price"]);
    sheet.setFrozenRows(1);
  }

  const targetCargo = String(data.cargoName).trim().toUpperCase();
  const newPrice = extractSafeNumber(data.basePrice);

  if (newPrice <= 0) {
    throw new Error("Invalid base price decimal provided. Must be greater than 0.");
  }

  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const cargoRange = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (let i = 0; i < cargoRange.length; i++) {
      if (String(cargoRange[i][0]).trim().toUpperCase() === targetCargo) {
        sheet.getRange(i + 2, 5).setValue(newPrice);
        return createJsonResponse({ status: "success", action: "UPDATED", cargo: targetCargo, price: newPrice });
      }
    }
  }

  sheet.appendRow([data.cargoName, "Custom Category", "Standard", "Base Game", newPrice]);
  return createJsonResponse({ status: "success", action: "INSERTED", cargo: data.cargoName, price: newPrice });
}

function handleRegisterUser(data) {
  const { sheet } = getTargetSheet("USERS_DB");
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["USERNAME", "PASSWORD", "DISCORD", "STEAM", "TMP", "ROLE", "STATUS", "JOIN_DATE", "END_DATE", "REASON", "TRACKER_NAME"]);
    sheet.setFrozenRows(1);
  }

  const values = sheet.getDataRange().getValues();
  const requestedUser = String(data.username).trim().toUpperCase();
  const requestedTracker = String(data.trackerName || data.username).trim().toUpperCase();

  for (let i = 1; i < values.length; i++) {
    if (String(values[i][0]).trim().toUpperCase() === requestedUser) {
      throw new Error("Username already allocated across network.");
    }
    if (values[i][10] && String(values[i][10]).trim().toUpperCase() === requestedTracker) {
      throw new Error("Tracker ID collision: Telemetry callsign is already reserved.");
    }
  }

  const timestamp = Utilities.formatDate(new Date(), "GMT+05:30", "yyyy-MM-dd");
  sheet.appendRow([
    data.username,
    data.password,
    data.discord || "",
    data.steamId || "",
    data.tmpId || "",
    "RIDER",
    "PENDING",
    timestamp,
    "",
    data.reason || "",
    data.trackerName || data.username
  ]);

  return createJsonResponse({ status: "success", message: "Application submitted for Command review." });
}

function handleLoginUser(data) {
  const { sheet } = getTargetSheet("USERS_DB");
  const values = sheet.getDataRange().getValues();
  
  if (data.username === "MASTER" && data.password === "TNTC@LEADER") {
    return createJsonResponse({
      status: "success",
      user: { username: "MASTER", trackerName: "MASTER", role: "leader", status: "ACTIVE" }
    });
  }

  const inputUser = String(data.username).trim().toUpperCase();
  const inputPass = String(data.password);

  for (let i = 1; i < values.length; i++) {
    if (String(values[i][0]).trim().toUpperCase() === inputUser) {
      if (String(values[i][1]) === inputPass) {
        const userStatus = String(values[i][6]).toUpperCase();
        if (userStatus === "PENDING") throw new Error("Clearance status: PENDING. Verification underway.");
        if (userStatus === "SUSPENDED" || userStatus === "LEFT") throw new Error("Security clearance REVOKED. Access terminated.");
        
        return createJsonResponse({
          status: "success",
          user: {
            username: values[i][0],
            trackerName: values[i][10] || values[i][0],
            role: String(values[i][5]).toLowerCase(),
            discord: values[i][2],
            joinDate: values[i][7]
          }
        });
      } else {
        throw new Error("Invalid password signature.");
      }
    }
  }
  throw new Error("Operative not found within network roster.");
}

function handleApproveUser(data) {
  const { sheet } = getTargetSheet("USERS_DB");
  const values = sheet.getDataRange().getValues();
  const targetUser = String(data.username).trim().toUpperCase();

  for (let i = 1; i < values.length; i++) {
    if (String(values[i][0]).trim().toUpperCase() === targetUser) {
      const today = Utilities.formatDate(new Date(), "GMT+05:30", "yyyy-MM-dd");
      sheet.getRange(i + 1, 6).setValue("RIDER");
      sheet.getRange(i + 1, 7).setValue("ACTIVE");
      sheet.getRange(i + 1, 8).setValue(today);

      const trackerName = values[i][10] || values[i][0];

      try {
        const { sheet: evSheet } = getTargetSheet("ENTRY SHEET");
        const headersRange = evSheet.getRange(1, 1, 5, evSheet.getLastColumn()).getValues();
        let headerRowIndex = 1;

        for (let r = 0; r < 5; r++) {
          if (String(headersRange[r][1]).toUpperCase().includes('DATE') || String(headersRange[r][2]).toUpperCase().includes('EVENT')) {
            headerRowIndex = r + 1;
            break;
          }
        }
        
        const currentHeaders = evSheet.getRange(headerRowIndex, 1, 1, evSheet.getLastColumn()).getValues()[0];
        let exists = false;
        for (let c = 6; c < currentHeaders.length; c++) {
          if (String(currentHeaders[c]).trim().toUpperCase() === trackerName.toUpperCase()) {
            exists = true;
            break;
          }
        }
        if (!exists) {
          const nextCol = evSheet.getLastColumn() + 1;
          evSheet.getRange(headerRowIndex, nextCol).setValue(trackerName);
        }
      } catch (err) {}

      return createJsonResponse({ status: "success", authorized_user: targetUser });
    }
  }
  throw new Error("Target user non-existent in authorization database.");
}

function handleManageUser(data) {
  const { sheet } = getTargetSheet("USERS_DB");
  const values = sheet.getDataRange().getValues();
  const targetUser = String(data.username).trim().toUpperCase();

  for (let i = 1; i < values.length; i++) {
    if (String(values[i][0]).trim().toUpperCase() === targetUser) {
      if (data.actionType === "CHANGE_ROLE") {
        sheet.getRange(i + 1, 6).setValue(data.newRole.toUpperCase());
      } else if (data.actionType === "SUSPEND") {
        sheet.getRange(i + 1, 7).setValue("SUSPENDED");
        sheet.getRange(i + 1, 9).setValue(new Date().toISOString());
      } else if (data.actionType === "LEFT") {
        sheet.getRange(i + 1, 7).setValue("LEFT");
        sheet.getRange(i + 1, 9).setValue(new Date().toISOString());
      }
      return createJsonResponse({ status: "success", managed: targetUser });
    }
  }
  throw new Error("Target operative not found.");
}

function handleAddEvent(data) {
  const { sheet } = getTargetSheet("ENTRY SHEET");
  const dataRange = sheet.getRange(1, 1, 5, sheet.getLastColumn()).getValues();
  let headerRow = [];
  let headerRowIndex = 1;

  for (let i = 0; i < 5; i++) {
    if (String(dataRange[i][1]).toUpperCase().includes('DATE') || String(dataRange[i][2]).toUpperCase().includes('EVENT')) {
      headerRow = dataRange[i];
      headerRowIndex = i + 1;
      break;
    }
  }
  if (headerRow.length === 0) headerRow = dataRange[0];

  const newRow = new Array(headerRow.length).fill("");
  newRow[0] = "";
  newRow[1] = data.date || "";
  newRow[2] = data.eventName || "";
  newRow[3] = data.link || "";
  newRow[4] = data.category || "";
  newRow[5] = data.imageLink || "";

  if (data.attendedDrivers && Array.isArray(data.attendedDrivers)) {
    const attendedSet = new Set(data.attendedDrivers.map(d => String(d).trim().toUpperCase()));
    for (let i = 6; i < headerRow.length; i++) {
      const driverName = String(headerRow[i]).toUpperCase().trim();
      if (driverName && driverName !== "UNKNOWN" && !driverName.includes('ATTENDANCE')) {
        newRow[i] = attendedSet.has(driverName);
      }
    }
  }

  const lastDataRow = sheet.getLastRow();
  const searchLimit = Math.max(lastDataRow, headerRowIndex + 1);
  const colC = sheet.getRange(1, 3, searchLimit, 1).getValues();
  let targetRow = lastDataRow + 1;

  for (let r = headerRowIndex; r < colC.length; r++) {
    if (String(colC[r][0]).trim() === "") {
      targetRow = r + 1;
      break;
    }
  }

  sheet.getRange(targetRow, 1, 1, newRow.length).setValues([newRow]);
  return createJsonResponse({ status: "success", row: targetRow });
}

function handleUpdateEvent(data) {
  const { sheet } = getTargetSheet("ENTRY SHEET");
  const dataRange = sheet.getRange(1, 1, 5, sheet.getLastColumn()).getValues();
  let headerRow = [];
  let headerRowIndex = 1;

  for (let i = 0; i < 5; i++) {
    if (String(dataRange[i][1]).toUpperCase().includes('DATE') || String(dataRange[i][2]).toUpperCase().includes('EVENT')) {
      headerRow = dataRange[i];
      headerRowIndex = i + 1;
      break;
    }
  }
  if (headerRow.length === 0) headerRow = dataRange[0];

  const lastDataRow = sheet.getLastRow();
  const searchLimit = Math.max(lastDataRow, headerRowIndex + 1);
  const colC = sheet.getRange(1, 3, searchLimit, 1).getValues();
  let targetRow = -1;

  for (let r = headerRowIndex; r < colC.length; r++) {
    if (String(colC[r][0]).trim() === String(data.originalName).trim()) {
      targetRow = r + 1;
      break;
    }
  }

  if (targetRow === -1) throw new Error("Target event record missing from database.");

  const updateRow = new Array(headerRow.length).fill("");
  updateRow[0] = "";
  updateRow[1] = data.date || "";
  updateRow[2] = data.eventName || "";
  updateRow[3] = data.link || "";
  updateRow[4] = data.category || "";
  updateRow[5] = data.imageLink || "";

  if (data.attendedDrivers && Array.isArray(data.attendedDrivers)) {
    const attendedSet = new Set(data.attendedDrivers.map(d => String(d).trim().toUpperCase()));
    for (let i = 6; i < headerRow.length; i++) {
      const driverName = String(headerRow[i]).toUpperCase().trim();
      if (driverName && driverName !== "UNKNOWN" && !driverName.includes('ATTENDANCE')) {
        updateRow[i] = attendedSet.has(driverName);
      }
    }
  }

  sheet.getRange(targetRow, 1, 1, updateRow.length).setValues([updateRow]);
  return createJsonResponse({ status: "success", updatedRow: targetRow });
}

function handleDeleteEvent(data) {
  const { sheet } = getTargetSheet("ENTRY SHEET");
  const lastDataRow = sheet.getLastRow();
  const colC = sheet.getRange(1, 3, lastDataRow, 1).getValues();

  for (let r = colC.length - 1; r >= 0; r--) {
    if (String(colC[r][0]).trim() === String(data.eventName).trim()) {
      sheet.deleteRow(r + 1);
      return createJsonResponse({ status: "success", purged: data.eventName });
    }
  }
  throw new Error("Event targeted for deletion does not exist.");
}

function handleCreateTour(data) {
  const { sheet: masterSheet, ss } = getTargetSheet("TOUR_MASTER");
  const tourName = data.tourName || `Tour_${Date.now()}`;
  
  if (ss.getSheetByName(tourName)) {
    throw new Error(`Campaign ID Conflict: '${tourName}' table already active.`);
  }

  let driverNames = [];
  const { sheet: jobSheet } = getTargetSheet("Live_Job_Logs");

  if (jobSheet && jobSheet.getLastRow() > 1) {
    const jobData = jobSheet.getRange(2, 1, jobSheet.getLastRow() - 1, 3).getValues();
    const driverMap = new Map();
    const thresholdDate = new Date();
    thresholdDate.setDate(thresholdDate.getDate() - 30);

    jobData.forEach(row => {
      const dateVal = new Date(row[0]);
      const name = String(row[2]).trim();
      if (name && name.toUpperCase() !== 'UNKNOWN' && !isNaN(dateVal.getTime())) {
        if (dateVal >= thresholdDate) {
          driverMap.set(name.toUpperCase(), name);
        }
      }
    });
    driverNames = Array.from(driverMap.values()).sort();
  }

  if (driverNames.length === 0) {

function autoCloseExpiredTours() {
  const { sheet } = getTargetSheet("TOUR_MASTER");
  const values = sheet.getDataRange().getValues();
  const today = new Date();
  today.setHours(0, 0, 0, 0);

      // Calculate and lock in the Vanguard

    }
  }
}

function calculateTourVanguard(tourName) {
  const DB_WEBSITE = "19ryICHTmtAp1dekGdF2wwq5b_oAKkgxxEEzLVApn4GQ";
  
  try {
    const { ss: tourSs } = getTargetSheet("TOUR_MASTER");
    const routeSheet = tourSs.getSheetByName(tourName);
    if (!routeSheet || routeSheet.getLastRow() < 2) return;

    const data = routeSheet.getDataRange().getValues();
    const headers = data[0];
    
    let maxCount = 0;
    let vanguardDriver = "TBD";
    
    // Drivers start at column H (index 7)
    for (let c = 7; c < headers.length; c++) {
      const driver = String(headers[c]).trim().toUpperCase();
      if (!driver || driver === "UNKNOWN" || driver.includes("ATTENDANCE")) continue;
      
      let count = 0;
      for (let r = 1; r < data.length; r++) {
        const val = String(data[r][c]).trim().toUpperCase();
        if (val === 'TRUE' || val === '1' || val === 'YES' || val === '✓') {
          count++;
        }
      }
      
      if (count > maxCount) {
        maxCount = count;
        vanguardDriver = driver;
      }
    }

    if (vanguardDriver !== "TBD" && maxCount > 0) {
      // Log to Website Achievers
      const websiteSs = SpreadsheetApp.openById(DB_WEBSITE);
      let achieverSheet = websiteSs.getSheetByName("Website_Achievers");
      if (!achieverSheet) {
        achieverSheet = websiteSs.insertSheet("Website_Achievers");
        achieverSheet.appendRow(["TIMESTAMP", "MONTH_YEAR", "CATEGORY", "DRIVER_NAME", "VALUE"]);
        achieverSheet.setFrozenRows(1);
      }

      const timestamp = new Date().toISOString();
      const currentMonthLabel = new Date().toLocaleString('en-US', { month: 'long', year: 'numeric' }).toUpperCase();
      
      // Append All-Time & Current Month records (Value holds the Tour Name)
      achieverSheet.appendRow([timestamp, "ALL-TIME", "TOUR_VANGUARD", vanguardDriver, tourName]);
      achieverSheet.appendRow([timestamp, currentMonthLabel, "TOUR_VANGUARD", vanguardDriver, tourName]);
    }
  } catch (e) {
    console.error("Vanguard Calculation Error: " + e.message);
  }
}

function getColumnLetter(colIndex) {
  let letter = '';
  while (colIndex > 0) {
    let temp = (colIndex - 1) % 26;
    letter = String.fromCharCode(temp + 65) + letter;
    colIndex = Math.floor((colIndex - temp - 1) / 26);
  }
  return letter;
}

function runDynamicEconomyCycle() {
  const DB_JOBS = "1e6Q0Bd6QPGHrAwbqqIuLgdVnIP8PomSFVSs_Kg4XK6Y";
  const CARGO_SPREADSHEET_ID = "1Cjg1S_l9FVgtEKE4ddrx3UtAroj7tB4jarfx2Yledrs";
  
  const MIN_PRICE = 1.50;
  const MAX_PRICE = 15.00;

  // 1. Calculate the 7-day cutoff date
  const now = new Date();
  const sevenDaysAgo = new Date(now.getTime() - (7 * 24 * 60 * 60 * 1000));

  // 2. Open Live_Job_Logs and read data
  const jobSs = SpreadsheetApp.openById(DB_JOBS);
  const jobSheet = jobSs.getSheetByName("Live_Job_Logs");
  if (!jobSheet) throw new Error("Could not find Live_Job_Logs sheet.");
  
  const jobData = jobSheet.getDataRange().getValues();
  const cargoCounts = new Map();

  // 3. Build the frequency map (skip header row)
  for (let i = 1; i < jobData.length; i++) {
    const timestampStr = jobData[i][0]; // Column A
    const cargoName = String(jobData[i][3] || "").trim().toUpperCase(); // Column D
    
    if (!cargoName) continue;

    const rowDate = new Date(timestampStr);
    
    // Only count if the date is valid and within the last 7 days
    if (!isNaN(rowDate.getTime()) && rowDate >= sevenDaysAgo) {
      cargoCounts.set(cargoName, (cargoCounts.get(cargoName) || 0) + 1);
    }
  }

  // 4. Open Cargo_Rates sheet
  const cargoSs = SpreadsheetApp.openById(CARGO_SPREADSHEET_ID);
  const rateSheet = cargoSs.getSheetByName("Cargo_Rates");
  if (!rateSheet) throw new Error("Could not find Cargo_Rates sheet.");

  const rateData = rateSheet.getDataRange().getValues();
  if (rateData.length <= 1) return; // Exit if only headers exist

  const newPrices = [];

  // 5. Process rate adjustments (skip header row)
  for (let i = 1; i < rateData.length; i++) {
    const cargoName = String(rateData[i][0]).trim().toUpperCase(); // Column A
    let currentPrice = parseFloat(rateData[i][4]); // Column E
    
    // Fallback if price is missing or invalid
    if (isNaN(currentPrice) || currentPrice <= 0) {
      currentPrice = 6.50; 
    }

    const count = cargoCounts.get(cargoName) || 0;
    let multiplier = 1.0;

    // Apply supply/demand logic
    if (count === 0) {
      multiplier = 1.15; // +15%
    } else if (count >= 1 && count <= 10) {
      multiplier = 1.05; // +5%
    } else if (count > 10 && count <= 25) {
      multiplier = 0.95; // -5%
    } else if (count > 25) {
      multiplier = 0.85; // -15%
    }

    let newPrice = currentPrice * multiplier;

    // Apply strict safety caps
    if (newPrice < MIN_PRICE) newPrice = MIN_PRICE;
    if (newPrice > MAX_PRICE) newPrice = MAX_PRICE;

    // Push as a 2D array row for bulk write
    newPrices.push([newPrice]);
  }

  // 6. Bulk overwrite Column E (Row 2 downwards)
  rateSheet.getRange(2, 5, newPrices.length, 1).setValues(newPrices);
}

function processMonthlyAchievers() {
  const DB_WEBSITE = "19ryICHTmtAp1dekGdF2wwq5b_oAKkgxxEEzLVApn4GQ";
  const DB_JOBS    = "1e6Q0Bd6QPGHrAwbqqIuLgdVnIP8PomSFVSs_Kg4XK6Y";
  const DB_EVENTS  = "1NhiUDmXIfy6BXnLcXhAv8Mmp6rmSKzvdI_eznlWbLsw";

  // 1. Determine Previous Month & Year
  const now = new Date();
  // Set to the 1st of the current month, then subtract 1 day to get into the previous month
  const prevMonthDate = new Date(now.getFullYear(), now.getMonth(), 0); 
  const targetMonth = prevMonthDate.getMonth(); // 0-11
  const targetYear = prevMonthDate.getFullYear();
  
  const monthNames = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
  const targetMonthLabel = `${monthNames[targetMonth]} ${targetYear}`.toUpperCase();

  // 2. Scan Job Logs for Distance King
  let topDistanceDriver = "TBD";
  let maxDistance = 0;
  
  try {
    const jobSs = SpreadsheetApp.openById(DB_JOBS);
    const jobSheet = jobSs.getSheetByName("Live_Job_Logs");
    if (jobSheet && jobSheet.getLastRow() > 1) {
      const jobData = jobSheet.getDataRange().getValues();
      const distanceMap = new Map();

      for (let i = 1; i < jobData.length; i++) {
        const rowDate = new Date(jobData[i][0]);
        if (isNaN(rowDate.getTime())) continue;

        if (rowDate.getMonth() === targetMonth && rowDate.getFullYear() === targetYear) {
          const driver = String(jobData[i][2]).trim().toUpperCase();
          const distStr = String(jobData[i][12]).replace(/[^0-9.]/g, '');
          const distance = parseFloat(distStr) || 0;

          if (driver && driver !== "UNKNOWN" && distance > 0) {
            distanceMap.set(driver, (distanceMap.get(driver) || 0) + distance);
          }
        }
      }

      // Find Max
      for (const [driver, distance] of distanceMap.entries()) {
        if (distance > maxDistance) {
          maxDistance = distance;
          topDistanceDriver = driver;
        }
      }
    }
  } catch (e) {
    console.error("Failed to parse Job Logs for Achievers: " + e.message);
  }

  // 3. Scan Event Records for Event Champion
  let topEventDriver = "TBD";
  let maxEvents = 0;

  try {
    const eventSs = SpreadsheetApp.openById(DB_EVENTS);
    const eventSheet = eventSs.getSheetByName("ENTRY SHEET");
    if (eventSheet && eventSheet.getLastRow() > 1) {
      const eventData = eventSheet.getDataRange().getValues();
      
      // Find Header Row
      let headerRowIndex = 0;
      for (let r = 0; r < Math.min(5, eventData.length); r++) {
        if (String(eventData[r][1]).toUpperCase().includes('DATE')) {
          headerRowIndex = r;
          break;
        }
      }
      
      const headers = eventData[headerRowIndex];
      const eventMap = new Map();

      for (let i = headerRowIndex + 1; i < eventData.length; i++) {
        const rowDate = new Date(eventData[i][1]);
        if (isNaN(rowDate.getTime())) continue;

        if (rowDate.getMonth() === targetMonth && rowDate.getFullYear() === targetYear) {
          // Check attendance columns (Starts at index 6)
          for (let c = 6; c < headers.length; c++) {
            const driver = String(headers[c]).trim().toUpperCase();
            if (!driver || driver === "UNKNOWN" || driver.includes("ATTENDANCE")) continue;

            const attended = String(eventData[i][c]).trim().toUpperCase();
            if (attended === "TRUE" || attended === "1" || attended === "YES") {
              eventMap.set(driver, (eventMap.get(driver) || 0) + 1);
            }
          }
        }
      }

      // Find Max
      for (const [driver, events] of eventMap.entries()) {
        if (events > maxEvents) {
          maxEvents = events;
          topEventDriver = driver;
        }
      }
    }
  } catch (e) {
    console.error("Failed to parse Events for Achievers: " + e.message);
  }

  // 4. Save to Website_Achievers
  const websiteSs = SpreadsheetApp.openById(DB_WEBSITE);
  let achieverSheet = websiteSs.getSheetByName("Website_Achievers");
  if (!achieverSheet) {
    achieverSheet = websiteSs.insertSheet("Website_Achievers");
    achieverSheet.appendRow(["TIMESTAMP", "MONTH_YEAR", "CATEGORY", "DRIVER_NAME", "VALUE"]);
    achieverSheet.setFrozenRows(1);
  }

  const timestamp = new Date().toISOString();
  
  // Append Distance King
  if (topDistanceDriver !== "TBD") {
    achieverSheet.appendRow([timestamp, targetMonthLabel, "DISTANCE_KING", topDistanceDriver, Math.floor(maxDistance)]);
  }
  
  // Append Event Champion
  if (topEventDriver !== "TBD") {
    achieverSheet.appendRow([timestamp, targetMonthLabel, "EVENT_CHAMPION", topEventDriver, maxEvents]);
  }
}

    achieverSheet.appendRow([timestamp, "ALL-TIME", "EVENT_CHAMPION", topEventDriver, maxEvents]);
  }
}
